rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    /*
      এই rules-এ ডাটা ব্যবহারের জন্য public access রাখা হয়েছে,
      কারণ বর্তমান app custom password ব্যবহার করে Firebase Authentication নয়।
      Production ব্যবহারের আগে Firebase Authentication + user-based rules চালু করুন।
    */
    match /students/{studentId} {
      allow read, write: if true;
    }
    match /collections/{collectionId} {
      allow read, write: if true;
    }
  }
}
